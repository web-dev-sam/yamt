```rust
// crates/range: pair folders for the sweep
pub fn pair_folders(range_dir: &Path) -> Vec<PathBuf>;                              // every <repo key>/<left>-<right>/
impl PairLock { pub fn try_hold(folder: &Path) -> io::Result<Option<PairLock>>; }  // None = an open window holds it

// crates/agent/src/lib.rs (package yamt-agent) — IPC types (serde tag "kind" PascalCase as the design's TS shows)
pub struct AgentRun(pub u32);
pub struct AgentCommand { pub program: String, pub args: Vec<String>, pub home_folders: Vec<String> }
pub enum AgentStartReply { Started { run: AgentRun }, NotFound { program: String, searched: Vec<String> },
                           SpawnFailed { program: String, message: String }, NoCheckout }
pub enum CheckoutReply { Ready { folder: String }, Failed { message: String } }
pub enum CheckoutLook { Missing, Clean, Changed { folder: String, files: Vec<String>, head_moved: bool },
                        Unreadable { folder: String, message: String } }
pub struct Pair { pub repo: PathBuf, pub folder: PathBuf, pub new_commit: String }
pub struct Workspace { /* pair, checkout: Option<PathBuf> (Some = complete), agent: Option<(AgentRun, Running)>, next_run */ }
impl Workspace {
    pub fn new(pair: Pair) -> Workspace;
    pub fn prepare(&mut self) -> CheckoutReply;                 // blocking; makes or reuses <pair>/checkout
    pub fn start(&mut self, command: &AgentCommand, events: Sender<HostEvent>) -> AgentStartReply; // NoCheckout unless prepared; a running agent stopped first; cwd = the checkout
    pub fn send(&self, run: AgentRun, message: String);
    pub fn stop(&self, run: AgentRun);
    pub fn stop_agent(&self);                                   // page reload
    pub fn look_job(&self) -> LookJob;                          // cheap, under the shell's lock
    pub fn discard(&mut self) -> Result<(), Error>;             // stop + reap the agent, then remove
    pub fn end(self);                                           // exit: stop + reap, then remove only when Clean
}
pub struct LookJob { /* folder, new commit */ }
impl LookJob { pub fn look(self) -> CheckoutLook; }            // git status outside the lock
pub fn sweep_leftovers(range_dir: &Path, own: &Path);
pub enum Error { Git(yamt_gitio::Error), Remove { folder: PathBuf, source: io::Error } }
```

## Change

As the design: making the checkout into `checkout.partial` (clone --shared --no-checkout, remove origin, checkout
--detach the right commit) then renaming to `checkout`; reusing a complete one (kept changes and commits included);
removing through `checkout.removing`; "changed" = any status entry or `HEAD` ≠ the right commit; a checkout git cannot
read → `Unreadable`, never removed by `end` or the sweep; the agent's program looked up with `find` (no project
entries: PATH, then the home entries the command gives); the sweep (other pairs whose lock `try_hold` takes:
`*.partial`/`*.removing` deleted, a clean `checkout` deleted, a changed one kept; its own pair: only
`.partial`/`.removing`); `agent-echo` (replies to lines, plus `pwd`, `edit <file>`, `stderr <text>`, `exit <code>`,
`huge`, `hold <file>`).

```rust
// crates/review/src/lib.rs (package yamt-review; #![forbid(unsafe_code)], #![warn(missing_docs)])
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize, Type)] pub struct ThreadId(pub u32);
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize, Type)] pub struct MessageId(pub u32);
#[serde(tag = "kind", rename_all = "camelCase")]
pub enum ThreadLine {
    Removed { left: LeftLine },
    Added { right: RightLine },
    UnchangedOnLeft { left: LeftLine, right: RightLine },
    UnchangedOnRight { left: LeftLine, right: RightLine },
}
#[serde(tag = "kind", rename_all = "camelCase")]
pub enum MessageBody { Note { text: String }, AiAnswer { turn: u32 } }
#[serde(tag = "kind", rename_all = "camelCase")]
pub enum Posting {
    NotQueued,
    Queued { text: String },
    Refused { text: String, reason: String },
    Pending { text: String, draft_note: u64, merge_request: String, link: String },
}
pub struct Message { pub id: MessageId, pub body: MessageBody, pub written: u64, pub posting: Posting }
pub struct Thread { pub id: ThreadId, pub file: RangeFileId, pub line: ThreadLine, pub messages: Vec<Message>, pub session: Option<String> }
pub struct ThreadList { pub threads: Vec<Thread> }
pub struct ThreadAdded { pub thread: ThreadId, pub list: ThreadList }

#[derive(Debug, thiserror::Error)]
pub enum Error {
    ReadFailed { path: PathBuf, message: String },
    Broken { path: PathBuf, message: String },      // serde's line and column where it has one; also repeated ids,
                                                    // an id at or above its counter, a thread without messages, a counter at u32::MAX
    NewerFormat { path: PathBuf, format: u32 },
    WriteFailed { path: PathBuf, message: String },
    UnknownFile { file: RangeFileId },
    UnknownThread { thread: ThreadId },
    UnknownMessage { message: MessageId },
    EmptyMessage,
    MessagePending { message: MessageId },          // queue or unqueue on a pending message
}

pub struct ThreadStore { /* private */ }
impl ThreadStore {
    pub fn open(folder: &Path) -> Result<ThreadStore, Error>;            // reads <folder>/threads.json; missing = empty, nothing written
    pub fn list(&self, range: &Range) -> ThreadList;                     // threads on listed paths only, in creation order
    pub fn add_thread(&mut self, range: &Range, file: RangeFileId, line: ThreadLine, text: &str) -> Result<ThreadAdded, Error>;
    pub fn add_note(&mut self, range: &Range, thread: ThreadId, text: &str) -> Result<ThreadList, Error>;
    pub fn queue(&mut self, range: &Range, message: MessageId, text: &str) -> Result<ThreadList, Error>;
    pub fn unqueue(&mut self, range: &Range, message: MessageId) -> Result<ThreadList, Error>;
}
```

## Change

1. `crates/range/src/lib.rs`: the three accessors (+ `entries`) and `Compared`.
2. `thread_types.rs`: the IPC types. `thread_file.rs`: format 1 exactly as the design's §3 JSON (`format`,
   `nextThread`, `nextMessage`, `threads[]` with `id`, `path`, `line`, `session`, `messages[]` with `id`, `body`,
   `written`, `posting`), every refusal of §3 "Reading" (the file left untouched), temp file + rename in the folder,
   a leftover temp file ignored. `thread_store.rs`: `ThreadStore` — a thread's path is the entry's `path` (stored);
   replies map it back to the entry's `RangeFileId`; threads on paths not in this range are kept on every write and
   left out of replies (one `warn` line naming them); `written` = seconds since 1970 stamped here; blank text (after
   trim) → `EmptyMessage`, nothing written; queue keeps the edited text (`Queued { text }`), a refused message queued
   again becomes `Queued`; unqueue → `NotQueued`; both refused on `Pending` with `MessagePending`; every change
   written before memory changes (a failed write changes nothing in memory). Keep `pub(crate)` hooks for row 16:
   `queued(&self) -> Vec<(ThreadId, MessageId, …)>` in thread then message order, and `record(message, Posting)`.
3. `arch-check.sh`: `yamt-review` depends on `yamt-range`, `yamt-gitio` and `yamt-merge-core` only among `yamt-*`,
   never on tauri; `yamt-app` may also depend on `yamt-review`.

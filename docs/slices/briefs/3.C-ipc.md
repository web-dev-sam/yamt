// ipc/compareDomain.ts — the compare window's domain shapes (camelCase); brands come from domain.ts guards
export type TabId = number & { readonly __brand: "TabId" };
export type ColumnVersion = number & { readonly __brand: "ColumnVersion" };
export type LeftLine = number & { readonly __brand: "LeftLine" };
export type RightLine = number & { readonly __brand: "RightLine" };
export type Column = "left" | "right";
export type TabLanguage =
  | { readonly kind: "auto" }
  | { readonly kind: "picked"; readonly name: string };
export type TabEntry = {
  readonly id: TabId;
  readonly name: string;
  readonly language: TabLanguage;
  readonly holdsText: boolean;
};
export type TabList = { readonly tabs: readonly TabEntry[]; readonly shown: TabId };
export type CompareStart = { readonly list: TabList; readonly columnLimit: number };
export type ColumnVersions = Readonly<Record<Column, ColumnVersion>>;
export type Difference = {
  readonly left: LineRange<LeftLine>;
  readonly right: LineRange<RightLine>;
  readonly words: { readonly left: SideWords<LeftLine>; readonly right: SideWords<RightLine> };
};
export type Comparison = {
  readonly versions: ColumnVersions;
  readonly differences: readonly Difference[];
};
export type TabTexts = {
  readonly left: string;
  readonly right: string;
  readonly comparison: Comparison;
};
export type ColumnWrite = {
  readonly tab: TabId;
  readonly column: Column;
  readonly text: string;
  readonly version: ColumnVersion;

## Behaviour

1. **Domain conversion** (`compareDomain.ts`, using guards and `toRange`/`toSideWords` that `domain.ts`/`words.ts`
   export for it; `domain.ts` stays ≤ 300 lines): every reply converted once, brands by guard, no `as` (TS4);
   `holds_text` → `holdsText`, `column_limit` → `columnLimit`.
2. **Tauri source** (`tauriCompare.ts`): one method per command through the existing `toResult`; the settings methods
   are the merge window's (`createTauriSettings`), not a copy.
3. **Fixture source** (`fixtureCompare.ts`, in memory, `createFixtureSettings()` for settings): starts with one empty
   "Compare 1"; `?raw=<case>` (a `fixtures/compare/<case>.json` name) puts that case's texts into tab 1; the tab rules
   as the facts list them (one implementation of each rule in this file); `setColumn` answers with the differences of a
   fixture case whose `left` and `right` equal the stored texts exactly, else no differences and one `info` log
   `compare.fixture_miss` (fields: tab, the two lengths); fixture JSON parsed with valibot (the one runtime parse, plan
   §2 #19), `import.meta.glob` literal as `fixture.ts` does it.

## Acceptance

- `compareDomain.test.ts`: a reply with every field converts (camelCase, brands, words both columns); a negative or
  fractional line/version/id is refused (the existing `invalid` path); `nextColumnVersion` adds one.
- `fixtureCompare.test.ts` (each naming the bug it catches): first start one "Compare 1"; `?raw=json-payload` puts the
  case's texts into tab 1 and `loadTab` returns its differences; numbers after closes (close "Compare 2" of three → next
  "Compare 4"); closing the shown tab shows the next, the last shown the previous, the last tab a fresh "Compare 1";
  rename trims, spaces only keeps; `setColumn` with a stale version stores nothing and replies with the stored
  versions; over the limit (a multi-byte text whose UTF-16 length is under it) → `ColumnTooLarge`; a miss logs
  `compare.fixture_miss` and answers no differences; unknown tab → `UnknownTab`; a broken fixture file fails the parse.
- `tauri.test.ts`/a new `tauriCompare.test.ts` only if an existing pattern tests the mapping beyond forwarding (no
  wiring tests).

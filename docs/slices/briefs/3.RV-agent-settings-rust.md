## Contract (fixed; you may add, never change)

```rust
// crates/settings/src/lib.rs
#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize, Type)]
#[serde(default, rename_all = "camelCase")]
pub struct AgentSettings {
    pub command: Option<Vec<String>>,       // None = the window's default (["omp", "acp"]); never an empty list on disk
    pub first_message: Option<String>,      // None = the window's default template
}
// StoredSettings gains `pub agent: AgentSettings` (a missing entry reads as the default; the entry is written only
// when a field is Some)
pub fn set_agent(places: &Places, current: &StoredSettings, agent: AgentSettings) -> Result<SettingsRead, Error>;
// src-tauri/src/settings.rs
#[tauri::command] #[specta::specta]
pub(crate) fn set_agent<R: Runtime>(…, current: StoredSettings, agent: AgentSettings) -> Result<SettingsRead, IpcError>;
```

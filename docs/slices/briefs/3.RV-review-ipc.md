## Contract (fixed; you may add, never change)

```ts
// ipc/domain.ts: ThreadId, MessageId brands by type guard (IPC3)
// ipc/threadLine.ts: toThreadLine(raw) / fromThreadLine(line) between the generated and the domain shape
// ipc/reviewDomain.ts: camelCase shapes (draftNote, mergeRequest, webUrl, lookedAt, shortId, gitlabHead, …) and
//   toThreadList, toThreadAdded, toGitLabCheck, toSendReply
// ipc/ReviewSource.ts
export interface ReviewSource {
  listThreads(): Promise<SourceResult<ThreadList>>;
  addThread(file: RangeFileId, line: ThreadLine, text: string): Promise<SourceResult<ThreadAdded>>;
  addNote(thread: ThreadId, text: string): Promise<SourceResult<ThreadList>>;
  queueMessage(message: MessageId, text: string): Promise<SourceResult<ThreadList>>;
  unqueueMessage(message: MessageId): Promise<SourceResult<ThreadList>>;
  checkGitLab(): Promise<SourceResult<GitLabCheck>>;
  sendQueue(onProgress: (progress: SendProgress) => void): Promise<SourceResult<SendReply>>;
}
// ipc/RangeSource.ts: interface RangeSource extends SettingsSource, ReviewSource { openRange; loadRangeFile }
// createTauriRangeSource / createFixtureRangeSource include the review methods (tauriReview.ts / fixtureReview.ts)
// fixture: threads in memory; ?threads=seeded (one thread of each ThreadLine kind on the review case, plus one queued,
// one refused, one pending and one aiAnswer message); ?gitlab=<script>: linedUp (default), checking (answers after
// 3 s), noGlab, loggedOut, noMergeRequest, noBranch, moved, otherBase, several, preparing, failed, refusal (the
// second queued message refused: "400 Bad request - Note {:line_code=>[\"can't be blank\"]}"), stopped (a timeout
// after the first); progress per message with a 300 ms pause
```

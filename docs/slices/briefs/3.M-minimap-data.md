## Contract (fixed; you may add, never change — W builds on these names)

```ts
// pane.ts — one syntax table; the pane's look unchanged
export type SyntaxToken = "--color-syntax-keyword" | /* … every --color-syntax-* token paletteHighlight uses */;
export const SYNTAX_COLOURS: readonly { readonly tag: Tag | readonly Tag[]; readonly colour: SyntaxToken }[];
export const syntaxColourClasses: Highlighter; // @lezer/highlight tagHighlighter: a tag's class is its SYNTAX_COLOURS token
// paletteHighlight is built from SYNTAX_COLOURS (color: var(<token>)) plus the existing non-colour styles.

// minimapLayout.ts — VS Code's proportional MinimapLayout in yamt's numbers; rows numbered from 1
export type MinimapInput = {
  readonly scrollTop: number; readonly viewportHeight: number;     // the Result's scroller, CSS px
  readonly paddingTop: number; readonly paddingBottom: number;     // the Result's documentPadding
  readonly rows: number;                                           // the Result's visual rows (a collapsed run = 1)
  readonly height: number;                                         // the minimap's CSS height
  readonly pixelRatio: number;
};
export type Slider = { readonly top: number; readonly height: number };        // CSS px in the minimap
export type MinimapLayout = {
  readonly input: MinimapInput; readonly startRow: number; readonly endRow: number;
  readonly topPaddingRows: number; readonly rowPx: number /* device px */;
  readonly slider: Slider | null /* null: the Result fits, nothing to drag */; readonly ratio: number;
};
export type Box = { readonly x: number; readonly width: number };               // device px from the minimap's left
export type Columns = { readonly left: Box; readonly code: Box; readonly right: Box; readonly width: number };
export function minimapScale(pixelRatio: number): 1 | 2;
export function minimapColumns(pixelRatio: number): Columns;   // 6 CSS px edge | 120 × scale device px code | 6 CSS px edge
export function minimapLayout(input: MinimapInput, previous: MinimapLayout | null): MinimapLayout;
export function rowAt(layout: MinimapLayout, offsetY: number): number;    // CSS px from the box top → row in [1, rows]
export function rowTop(layout: MinimapLayout, row: number): number;       // device-px y of that row's top in the canvas
export function centredScrollTop(row: number, input: MinimapInput): number;     // VS Code Center
export function draggedScrollTop(start: MinimapLayout, delta: number): number;  // delta: CSS px moved since the press

// minimapGlyphs.ts — VS Code's prebaked tables, MIT header naming Microsoft Corporation and the source files
export function glyphs(scale: 1 | 2): Uint8ClampedArray;  // intensity per glyph pixel, 96 glyphs
export function charIndex(code: number): number;            // VS Code getCharIndex

// minimapPixels.ts — writes RGBA into an ImageData-shaped buffer; no DOM
export type Rgb = { readonly r: number; readonly g: number; readonly b: number };
export type PixelRows = { readonly data: Uint8ClampedArray; readonly width: number };  // width in device px
export type CodeRow = { readonly y: number; readonly x: number; readonly text: string;
  readonly colours: Uint8Array /* palette index per character of text */; readonly tabSize: number };
export type GlyphLook = { readonly scale: 1 | 2; readonly light: boolean; readonly palette: readonly Rgb[] };
export function rgbOf(value: string): Rgb | null;            // "#rrggbb" or "#rrggbbaa" (alpha ignored), else null
export function paintRow(target: PixelRows, row: CodeRow, look: GlyphLook): void; // RGB of the token, alpha = intensity
export function paintWave(target: PixelRows, wave: { readonly y: number; readonly x: number; readonly width: number;
  readonly rowPx: number }, colour: Rgb): void;              // one device px line across the row's middle

// minimapTiles.ts — the staleness rule only (row 3 adds the canvas cache class here)
export const TILE_ROWS = 128;
export type Stale = "none" | "all" | { readonly from: number };   // from: document position
export function staleTiles(facts: { readonly docChanged: boolean; readonly foldsChanged: boolean;
  readonly themeChanged: boolean; readonly languageChanged: boolean;
  readonly treeBefore: number; readonly treeAfter: number /* syntax tree lengths */ }): Stale;

// minimapMarks.ts — the two edges
export type EdgeKind = Tone | "error" | "warning";
export type EdgeMark = { readonly top: number; readonly bottom: number; readonly kind: EdgeKind };  // Result document px
export type MarkBox = { readonly x: number; readonly y: number; readonly width: number; readonly height: number;
  readonly kind: EdgeKind };                                           // device px in the canvas
export function changeMarks(rows: readonly ChunkRow[]): EdgeMark[];  // row.center, paneTone(…, "center"); null → none
export function problemMarks(state: EditorState, blockAt: (pos: number) => Span): EdgeMark[]; // error, warning only; warnings first
export function markBoxes(marks: readonly EdgeMark[], layout: MinimapLayout, edge: Box): MarkBox[];
// markBoxes: 4 CSS px wide, 1 CSS px in from the edge's left (device-px rounded); never under 4 CSS px tall, a shorter
// span (an insertion point too) centred on it; clamped inside the canvas; only marks inside the window.
```

## Change

1. `pane.ts`: the `SYNTAX_COLOURS` table, `paletteHighlight` built from it, `syntaxColourClasses`.
2. `minimapLayout.ts`: VS Code's `MinimapLayout.create` (proportional, no whitespace gaps, `scrollBeyondLastLine` off),
   `minimapScale`, `minimapColumns`, `rowAt`, `rowTop`, `centredScrollTop`, `draggedScrollTop`.
3. `minimapGlyphs.ts`: VS Code's prebaked data for both scales, decoded once, and `charIndex`.
4. `minimapPixels.ts`: `rgbOf`, `paintRow` (glyph intensities ×12/15 dark, ×50/60 light; tabs, spaces, 120-column cut;
   characters outside 32–126 through `charIndex`), `paintWave`.
5. `minimapTiles.ts`: `TILE_ROWS`, `staleTiles` (document, folds, theme or language change → `"all"`; the tree grown
   without those → `{ from: treeBefore }`; otherwise `"none"`; statuses never stale tiles).
6. `minimapMarks.ts`: `changeMarks`, `problemMarks`, `markBoxes`.

## Contract (fixed; you may add, never change)

```ts
// ui/rangeThreads.ts
export type RangeThreads = {
  readonly list: Readonly<ShallowRef<ThreadList>>;              // mirrors the last reply
  readonly open: Readonly<ShallowRef<ReadonlySet<ThreadId>>>;   // never stored; reopened ranges start collapsed
  readonly drafts: Readonly<ShallowRef<ReadonlyMap<string, Draft>>>; // new-thread drafts and reply text, kept across the 16-view eviction
  readonly counts: ComputedRef<ReadonlyMap<RangeFileId, number>>;   // per file, drafts excluded
  shownFor(file: RangeFileId): readonly ShownThread[];
  startThread(file: RangeFileId, line: ThreadLine): void;       // opens a draft (a second press focuses it)
  toggle(keys: readonly string[]): void;
  addThread(draftKey: string, text: string): Promise<void>;     // the draft's key becomes the stored id, text kept on failure
  addNote(thread: ThreadId, text: string): Promise<void>;
  queue(message: MessageId, text: string): Promise<void>;
  unqueue(message: MessageId): Promise<void>;
  replace(list: ThreadList): void;                               // after a send
};
export function createRangeThreads(source: ReviewSource, report: (problem: Problem) => void): RangeThreads;
// ui/rangeGitLab.ts
export type GitLabState = {
  readonly check: Readonly<ShallowRef<GitLabCheck | null>>;     // null until the first answer
  readonly checking: Readonly<ShallowRef<boolean>>;
  readonly sending: Readonly<ShallowRef<SendProgress | null>>;
  checkNow(): Promise<void>;                                     // one at a time
  checkOnFocus(): void;                                          // when the last check ended ≥ RECHECK_PAUSE_MS (30 s) ago
  send(): Promise<void>;                                         // SendReply → threads.replace, notice "5 sent, 1 refused"
};
export function createGitLabCheck(source: ReviewSource, threads: RangeThreads, report: …): GitLabState;
// ui/rangeThreadText.ts: every sentence — each NotLinedUp (exhaustive), the posting labels ("Queued for GitLab",
//   "Refused by GitLab: …", "Pending on GitLab"), "Checking GitLab…", "Nothing queued", the Send label
//   "Send to GitLab (n)" / "Sending 3 of 7…", the Send and Post hovers, threadPlaceText(entry, line) ("src/board.ts:12",
//   the old path and line for a removed line; also C's RELATIVE_FILE_PATH_WITH_LINE), the two-dots hint
// rangeStore gains `threads: RangeThreads` and `gitlab: GitLabState`; checkNow at start after the list shows
```

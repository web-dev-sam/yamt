// ui/settingsStore.ts — the settings part any window root offers the settings view (plan §2 #94)
export type SettingsStore = Pick<
  SettingsActions,
  | "setTheme"
  | "setAntialiasing"
  | "moveSettings"
  | "addKey"
  | "changeKey"
  | "removeKey"
  | "removeAllKeys"
  | "resetKeys"
> & {
  readonly settings: Readonly<ShallowRef<SettingsInUse>>;
  openSettings(): void;
  closeSettings(): void;
};
export function registerSettingsStore(store: SettingsStore): void; // the root's store calls it once at creation
export function useSettingsStore(): SettingsStore; // throws before registerSettingsStore

// ui/noticeText.ts
export type NoticeRow = {
  readonly key: string;
  readonly text: string;
  readonly closeLabel: string;
};

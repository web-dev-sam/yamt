import { trackEvent } from "./analytics";
import { formatMoney } from "./money";
import type { Product } from "./catalog";

export const MAX_QUANTITY = 99;
const TAX_RATE = 0.2;

export interface CartLine {
  readonly product: Product;
  quantity: number;
}

export interface Cart {
  readonly id: string;
  lines: CartLine[];
  couponCode: string | null;
}

export function createCart(id: string): Cart {
  return { id, lines: [], couponCode: null };
}

export function addItem(cart: Cart, product: Product, quantity = 1): Cart {
  const existing = cart.lines.find((line) => line.product.id === product.id);
  if (existing) {
    existing.quantity = Math.min(existing.quantity + quantity, MAX_QUANTITY);
    return cart;
  }
  cart.lines.push({ product, quantity });
  return cart;
}

export function removeItem(cart: Cart, productId: string): Cart {
  const before = cart.lines.length;
  cart.lines = cart.lines.filter((line) => line.product.id !== productId);
  if (cart.lines.length < before) {
    trackEvent("cart.remove", { cartId: cart.id, productId });
  }
  return cart;
}

export function setQuantity(cart: Cart, productId: string, quantity: number): Cart {
  const line = cart.lines.find((candidate) => candidate.product.id === productId);
  if (line) {
    line.quantity = Math.max(0, Math.min(quantity, MAX_QUANTITY));
  }
  return cart;
}

export function subtotal(cart: Cart): number {
  let sum = 0;
  for (const line of cart.lines) {
    sum += line.product.price * line.quantity;
  }
  return sum;
}

export function discount(cart: Cart): number {
  if (cart.couponCode === "WELCOME10") {
    return Math.min(subtotal(cart) * 0.1, 50);
  }
  return 0;
}

export function tax(cart: Cart): number {
  return (subtotal(cart) - discount(cart)) * TAX_RATE;
}

export function total(cart: Cart): number {
  return subtotal(cart) - discount(cart) + tax(cart);
}

export function summary(cart: Cart): string {
  const count = cart.lines.reduce((n, line) => n + line.quantity, 0);
  return `${count} items, ${formatMoney(total(cart))}`;
}

export function isEmpty(cart: Cart): boolean {
  return cart.lines.every((line) => line.quantity === 0);
}
